function smartTimes(){let vals=roads.map(r=>demand[r.id]),sum=vals.reduce((a,b)=>a+b,0);let t=vals.map(v=>Math.max(15,Math.min(45,120*v/sum)));for(let j=0;j<5;j++){let s=t.reduce((a,b)=>a+b,0);if(Math.abs(s-120)<.1)break;let free=t.map((x,i)=>({i,free:s>120?x>15:x<45})).filter(x=>x.free);if(!free.length)break;let diff=(120-s)/free.length;free.forEach(o=>t[o.i]=Math.max(15,Math.min(45,t[o.i]+diff)))}return t.map(x=>Math.round(x))}

}
